// Extraction de la recherche d'un PID libre
pid_t findFreePID() {
    for (int i = 0; i < MAX_PROCESS; i++) {
        if (process_table[i].state == TERMINE) {
            return i;
        }
    }
    return -1;
}

// Extraction de l'initialisation du contexte du processus
void initProcessContext(pid_t pid, const char *processName, void (*function)()) {
    process_table[pid].pid = pid;
    process_table[pid].name = processName;
    process_table[pid].state = PRET_ACTIF;
    process_table[pid].stack_top = STACK_SIZE - 1;
    process_table[pid].function = function;
    for (int i = 0; i < 5; i++) {
        process_table[pid].regs[i] = 0;
    }
    // Initialisation du registre esp
    process_table[pid].regs[1] = (uint32_t)&process_table[pid].stack[STACK_SIZE - 1];
}

pid_t creer(const char *processName, void (*function)()) {
    pid_t pid = findFreePID();
    if (pid == -1) {
        printf("Erreur : impossible de créer un nouveau processus\n");
        return -1;
    }

    initProcessContext(pid, processName, function);

    // Ajout à la file des processus prêts
    addProcess(pid);

    return pid;
}
